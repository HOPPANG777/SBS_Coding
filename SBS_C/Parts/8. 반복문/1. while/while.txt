	2.	while..
	-----------

		----------
		1.	형식..
		----------

			while( 반복 조건 )
			{
				반복 구간..

				....

				반복조건을 무너뜨리는 연산
				또는 카운팅 연산..

			}

	
			예)	
				int num = 0;
	
				while( num < 3 )
				{	
					printf("Hello !!\n");

					++num;
				}
