		-----------
		1.	구조체..
		-----------
		
			-	하나 이상의 변수등을 묶어서
				( 포인터 변수와 배열등을 포함.. )
				새로운 자료형을 정의..

			-	구조체 형식 선언..

				struct	구조체_이름
				{
					자료형	멤버변수1;
					자료형	멤버변수2;
					........
				};



			예)	마우스의 좌표정보를 관리해야 할때

					int xPos;
					int yPos;

				두 개의 변수를 선언해야함..

				이 두 변수는 항상 함께해야함..

				ㄴ	서로 독립된 정보를 표현하는게 아니라,
					[ 마우스의 위치 ]라는 하나의 정보를
					표현하기 때문..

				ㄴ	두 개 이상의 논리적으로 그룹화된 정보를
					하나의 정보로 표현하는 것이라면
					요소들을 하나로 묶어서 관리하는 것이
					효율적임..

					-	구조체 사용..

						//	좌표 정보..
						struct SPoint
						{
						   int _xPos;
						   int _yPos;
						};


						//	한 사람의 정보..
						struct SPerson
						{
						   char _name[20];		// 이름 표시
						   char _phoneNum[20];	// 전화번호	
						   int _age;			// 나이
						};
					

	----------------
	2.	구조체 변수의
		선언과 접근..
	----------------

		--------------
		1.	변수의
			선언 방법..
		--------------

			-	struct	구조체타입  객체이름;

				예)	struct SPoint pos;

					struct SPerson man;


		--------------
		2.	멤버변수의
			접근 방법..
		--------------
	
			-	구조체_변수_이름.멤버변수

				예)	pos._xPos = 10;
					pos._yPos = 20;

					strcpy(man._name,"aaa");
					strcpy(man._phoneNum,"010....");
					man._age = 20;

		(소스 참고)
		




	----------------
	3.	구조체 변수의
		초기화..
	----------------

		1.	구조체 변수의 초기화 방법은
			배열의 초기화와 유사..

			-	멤버의 순서대로
				초기값을 나열..

				ㄴ	초기화 과정에서 문자열 멤버변수의
					저장을 위해 strcpy등의 함수를
					사용하지 않아도 됨..

			예)

				int main()
				{
				   struct SPoint pos = { 10, 20 };

    			   struct SPerson man = { "홍길동", "010-1234-5678", 20 };

				   .....	

				   return 0;	

				}

				(소스 참고)



		
	------------------
	4.	구조체와
		typedef 선언..
	------------------

		----------
		1.	기능..
		----------

			-	특정한 타입에
				또다른 이름을 부여..
			
				ㄴ	빈번하게 사용되는 소스코드가
					복잡하거나 문장이 길 경우
					이를 간결하게 하는데 주로 사용..

				예)	typedef int INT;

					ㄴ	int에 INT라는 이름을
						추가로 붙여줌..

			-	새로운 이름의 부여는
				가장 마지막에 등장하는 단어를
				중심으로 적용..

				형식)	typedef type1 type2 type3 ... name;

						예)	typedef unsigned char * PTR_UCHAR;

							ㄴ	[ unsigned char * ]에
								[ PTR_UCHAR ]라는 이름을
								추가로 적용..


			-	[ typedef ]로 정의되는 자료형의 이름은
				[ 대문자 ]로 [ 시작 ]하는 것이 관례..

				ㄴ	기존 자료형의 이름과
					구분하기 위해..
			

		-----------
		2.	구조체에
			적용..
		-----------

			1)	구조체 선언후 typedef..

				예)
					struct SPoint
					{
						int _xPos;
						int _yPOs;
					};
		
					before..
					-	struct SPoint pos;

					typedef struct SPoint SPos;

					after..
					-	SPos pos;



			2)	구조체 정의와 typedef 동시 적용..

				예)
					typedef struct SPoint
					{
						int _xPos;
						int _yPOs;

					} SPos;


		
			3)	구조체 이름 생략..

				예)
					typedef struct
					{
						int _xPos;
						int _yPOs;

					}SPos;
