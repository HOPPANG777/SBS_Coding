	-----------------
	1.	리터럴 상수..
	-----------------

		----------
		-	상수..
		----------

			ㄴ	변경이 불가능한 데이터..
			
			ㄴ	데이터 자체..

				예)	12, 23.5f, "test", 'a',...
		


		-----------------
		-	리터럴 상수..
		-----------------

			-	일반적으로 변수에 할당되는
				데이터 자체..			

			-	예)	int num1 = 3 + 4;

					int num2 = 7 + num1;

					double num3 = 2.5 + 7.5;

					char ch1 = 'A';




		------------------------
		-	다양한 상수의 표현..
		------------------------

			-	일반적으로 정수형 상수는
				int형, 실수형 상수는 double형으로
				관리하기로 규정..

				( "1.	Literal Size" 참고 )


			-	하지만 정수형은 int만 있는 것도 아니고,
				( char, short, long 등 )
				실수형은 double만 있는것도 아님
				( float, long double 등 )


			-	float num1 = 3.14;

				->	double( 8바이트 )형 데이터 3.14를
					float( 4바이트 )형에
					저장하였으므로
					데이터 소실 가능..

					->	float형 상수로 관리해야 함..

					->	상수 뒤에 float형 상수라는
						의미로 f를 추가..

					->	float num1 = 3.14f;

						ㄴ	대소문자 구분하지 않음..
							
							ㄴ	3.14F도 가능..

			

			-	종류

				======================================================================
				타입	|	접미사	|	자료형			|	예						
				======================================================================
						|			|	int				|	int num = 1025;
				정수형	|	U		|	unsigned int	|	unsigned int num = 1025u;
						|	L		|	long			|	long num = 1025L;
						|	UL		|	unsigned long	|	unsigned long num = 1025L;
						|	LL		|	long long		|	long long num = 1025LL;
				======================================================================
						|			|	double			|	double num = 3.14;
				실수형	|	F		|	float			|	float num = 3.14f;
						|	L		|	long double		|	long double num = 5.78L;
				======================================================================





	-----------------
	2.	심볼릭 상수..
	-----------------

		--------------------
		-	const, #define..
		--------------------
			----------
			-	정의..
			----------

				ㄴ	이름을 지니는 상수..

					ㄴ	변수의 초기화와 유사..


			---------------
			-	표현 방식..
			---------------

				const int BULLET_COUNT = 10;

				#define BULLET_COUNT  10


				->	BULLET_COUNT는 컴파일 할때
					10이라는 데이터와 동일하게
					처리됨..


				//	변수..
					int a = 100;
					a = 30;

				//	상수..
					BULLET_COUNT = 100;	-	(X)

					ㄴ	10 = 100;



			------------------------
			-	심볼릭 상수의 장점..
			------------------------

				1)	가독성..

				2)	연산 속도..

					-	변수 < 상수..



			---------------
			-	주의 사항..
			---------------

				-	반드시 초기화할 것..

					ㄴ	const int BULLET_COUNT;		-	( X )


				-	심볼릭 상수도 일종의 상수이므로
					초기화가 되면 값 변경 불가..

					예)	const int BULLET_COUNT = 10;
						
						BULLET_COUNT = 100;
						
						ㄴ	10 = 100;


				-	상수 작명시 관례..

					1)	상수의 이름은 모두
						[ 대문자 ]로 설정..

					2)	둘 이상의 단어를 결합할때						
						"_""를 이용하여 두 단어를 구분..

						예)	BULLET_COUNT..


				-	가급적이면 const 사용 권장..

					ㄴ	#define은 디버깅할때
						추적이 되지 않음..


				( "2.	심볼릭 상수" 참고 )



		-----------------
		-	열거형 상수..
			[ 참고	:	https://blog.naver.com/josep0302/222178670736 
						https://blog.naver.com/coms1101/222304351219 ]
		-----------------
		
			----------
			-	정의..
			----------

				ㄴ	심볼릭 상수를 열거한
					그룹화 된 상수..
		

			----------
			-	형식..
			----------

				enum	열거형 이름
				{
					열거자_1	=	초기값,
					열거자_2,
					열거자_3,

					....
				}



			----------
			-	특징..
			----------

				-	열거자는 별도로
					초기화 가능..

				-	초기화를 하지 않으면
					첫번째 요소 [ 열거자_1 ]은
					0 으로 자동 세팅..
				
				-	열거자의 값은 연속적으로
					1씩 증가됨..

				-	열거자의 값을 특정 값으로
					바꾸면 그 이후에 값들은
					갱신한 값을 기준으로
					1씩 증가..

					예)
						enum eNATION
						{
							KOREA,
							AMERAICA,
							JAPAN,
							CHINA = 100,
							INDIA,	--->	INDIA = 101, SWISS = 102...
							SWISS,
							...
						}


				-	열거형으로 변수를 생성 가능..
					
					ㄴ	정수형 변수 관리 가능..


			( "3.	열거형 상수 ( enum )" 참고 )
